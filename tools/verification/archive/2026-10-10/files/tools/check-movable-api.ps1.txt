# Compile real downstream consumers; expected failures require exact diagnostics.
param([string]$Stage = 'target/movable-api-controls')
$ErrorActionPreference = 'Stop'
$taskRoot = [IO.Path]::GetFullPath((Join-Path $PSScriptRoot '..'))
$stagePath = [IO.Path]::GetFullPath((Join-Path $taskRoot $Stage))
New-Item -ItemType Directory -Force (Join-Path $stagePath 'src') | Out-Null
$wrapper = (Join-Path $taskRoot 'crates/fsr-sdk').Replace('\', '/')
$raw = (Join-Path $taskRoot 'crates/fsr-sdk-sys').Replace('\', '/')
$manifest = @"
[package]
name = "movable-api-control"
version = "0.0.0"
edition = "2024"
[workspace]
[features]
dx12 = ["fsr-sdk/dx12"]
movable = ["fsr-sdk/movable"]
[dependencies.fsr-sdk]
path = "$wrapper"
default-features = false
[dependencies.fsr-sdk-sys]
path = "$raw"
default-features = false
"@
[IO.File]::WriteAllText((Join-Path $stagePath 'Cargo.toml'), $manifest)
$cases = @(
    @{ Name='common-positive'; Features=''; Diagnostic=$null; Source='use fsr_sdk::upscaler::{Dimensions, common}; use fsr_sdk_sys::{api::ffxApiHeader, upscale::ffxDispatchDescUpscale}; fn main() { let _: Dimensions = common::Dimensions::new(320, 180).unwrap(); let _: Option<ffxApiHeader> = None; let _: Option<ffxDispatchDescUpscale> = None; }' },
    @{ Name='common-movable-positive'; Features='movable'; Diagnostic=$null; Source='use fsr_sdk::upscaler::{CameraParameters, FrameTimeMillis, JitterOffsetPixels, UpscalerOptions, common}; fn main() { let d = common::Dimensions::new(320, 180).unwrap(); let _: UpscalerOptions = common::UpscalerOptions::new(d, d); let _: CameraParameters = common::CameraParameters::new(0.1, 100.0, 1.0, 1.0).unwrap(); let _: FrameTimeMillis = common::FrameTimeMillis::new(16.0).unwrap(); let _: JitterOffsetPixels = common::JitterOffsetPixels::new(0.0, 0.0).unwrap(); }' }
)
# Both configurations omit DX12 on Windows; even enabled native features must
# expose no native API on other hosts. Check each boundary independently.
$disabledFeatures = @('', 'movable')
if (-not $IsWindows) { $disabledFeatures += @('dx12', 'dx12,movable') }
foreach ($features in $disabledFeatures) {
    foreach ($boundary in @(
        @{ Name='runtime'; Import='fsr_sdk::runtime' },
        @{ Name='dx12-module'; Import='fsr_sdk::upscaler::dx12' },
        @{ Name='legacy-upscaler'; Import='fsr_sdk::upscaler::Upscaler' },
        @{ Name='dispatch-resources'; Import='fsr_sdk::upscaler::UpscaleResources' },
        @{ Name='native-sharpness'; Import='fsr_sdk::upscaler::NativeSharpness' },
        @{ Name='sharpening'; Import='fsr_sdk::upscaler::Sharpening' },
        @{ Name='legacy-movable'; Import='fsr_sdk::upscaler::MovableUpscaler' },
        @{ Name='raw-loader'; Import='fsr_sdk_sys::loader' },
        @{ Name='raw-dx12'; Import='fsr_sdk_sys::dx12' }
    )) {
        $cases += @{ Name="native-disabled-$($boundary.Name)-$($features.Replace(',', '-'))"; Features=$features; Diagnostic='E0432'; Source="use $($boundary.Import); fn main() {}" }
    }
}
if ($IsWindows) { $cases += @(
    @{ Name='dispatch-inputs-positive'; Features='dx12'; Diagnostic=$null; Source='use fsr_sdk::upscaler::{CameraParameters, FrameTimeMillis, JitterOffsetPixels, ID3D12GraphicsCommandList, ID3D12Resource, NativeSharpness, Sharpening, UpscaleDispatch, UpscaleResources, dx12}; fn frame<''a>(list: &''a ID3D12GraphicsCommandList, r: [&''a ID3D12Resource; 5]) -> UpscaleDispatch<''a> { let resources = UpscaleResources::new(r[0], r[1], r[2], r[3], r[4]); let copy: dx12::UpscaleResources = resources.clone(); let mut f = UpscaleDispatch::new(list, copy, JitterOffsetPixels::new(0.0, 0.0).unwrap(), FrameTimeMillis::new(16.0).unwrap(), CameraParameters::new(0.1, 100.0, 1.0, 1.0).unwrap(), true); let _: &Sharpening = f.sharpening(); f.set_sharpening(Sharpening::Native(NativeSharpness::new(0.35).unwrap())); match f.sharpening() { Sharpening::Disabled => {}, Sharpening::Native(v) => { let _ = v.value(); }, _ => {} } f } fn main() {}' },
    @{ Name='dispatch-literal-private'; Features='dx12'; Diagnostic='E0451'; Source='use fsr_sdk::upscaler::UpscaleDispatch; fn wrong(f: UpscaleDispatch) { let _ = UpscaleDispatch { ..f }; } fn main() {}' },
    @{ Name='dispatch-field-private'; Features='dx12'; Diagnostic='E0616'; Source='use fsr_sdk::upscaler::UpscaleDispatch; fn wrong(f: UpscaleDispatch) { let _ = f.reset; } fn main() {}' },
    @{ Name='resources-literal-private'; Features='dx12'; Diagnostic='E0451'; Source='use fsr_sdk::upscaler::UpscaleResources; fn wrong(r: UpscaleResources) { let _ = UpscaleResources { ..r }; } fn main() {}' },
    @{ Name='sharpness-construction-private'; Features='dx12'; Diagnostic='E0423'; Source='use fsr_sdk::upscaler::NativeSharpness; fn main() { let _ = NativeSharpness(0.35); }' },
    @{ Name='raw-loader-positive'; Features='dx12'; Diagnostic=$null; Source='use fsr_sdk_sys::loader::{FfxLibrary, LoadError}; fn main() { let _: Option<FfxLibrary> = None; let _: Option<LoadError> = None; }' },
    @{ Name='local-positive'; Features='dx12'; Diagnostic=$null; Source='use fsr_sdk::runtime::Runtime; fn main() { let _: Option<Runtime> = None; }' },
    @{ Name='movable-positive'; Features='dx12,movable'; Diagnostic=$null; Source='use fsr_sdk::{runtime::MovableRuntime, upscaler::MovableUpscaler}; fn send<T: Send>() {} fn main() { send::<MovableRuntime>(); send::<MovableUpscaler>(); }' },
    @{ Name='feature-disabled'; Features='dx12'; Diagnostic='E0432'; Source='use fsr_sdk::runtime::MovableRuntime; fn main() { let _: Option<MovableRuntime> = None; }' },
    @{ Name='dx12-disabled'; Features='movable'; Diagnostic='E0432'; Source='use fsr_sdk::runtime::MovableRuntime; fn main() { let _: Option<MovableRuntime> = None; }' },
    @{ Name='local-to-movable-construction'; Features='dx12,movable'; Diagnostic='E0308'; Source='use fsr_sdk::{runtime::Runtime, upscaler::{MovableUpscaler, UpscalerOptions, ID3D12Device}}; fn wrong(r: &Runtime, d: &ID3D12Device, o: UpscalerOptions) { let _ = MovableUpscaler::new(r, d, o); } fn main() {}' },
    @{ Name='movable-to-local-construction'; Features='dx12,movable'; Diagnostic='E0308'; Source='use fsr_sdk::{runtime::MovableRuntime, upscaler::{Upscaler, UpscalerOptions, ID3D12Device}}; fn wrong(r: &MovableRuntime, d: &ID3D12Device, o: UpscalerOptions) { let _ = Upscaler::new(r, d, o); } fn main() {}' },
    @{ Name='no-local-to-movable-conversion'; Features='dx12,movable'; Diagnostic='E0277'; Source='use fsr_sdk::runtime::{Runtime, MovableRuntime}; fn wrong(r: Runtime) -> MovableRuntime { r.into() } fn main() {}' },
    @{ Name='no-movable-to-local-conversion'; Features='dx12,movable'; Diagnostic='E0277'; Source='use fsr_sdk::runtime::{Runtime, MovableRuntime}; fn wrong(r: MovableRuntime) -> Runtime { r.into() } fn main() {}' }
) }
foreach ($case in $cases) {
    [IO.File]::WriteAllText((Join-Path $stagePath 'src/main.rs'), $case.Source)
    $cargoArgs = @('check', '--manifest-path', (Join-Path $stagePath 'Cargo.toml'), '--target-dir', (Join-Path $stagePath 'build'))
    if ($case.Features) { $cargoArgs += @('--features', $case.Features) }
    $output = & cargo @cargoArgs 2>&1 | Out-String
    $code = $LASTEXITCODE
    [IO.File]::WriteAllText((Join-Path $stagePath "$($case.Name).txt"), $output)
    if ($case.Diagnostic) {
        if ($code -eq 0 -or !$output.Contains("error[$($case.Diagnostic)]")) {
            throw "Invalid negative control $($case.Name): exit=$code`n$output"
        }
    } elseif ($code -ne 0) { throw "Positive control failed $($case.Name):`n$output" }
    Write-Host "$($case.Name): PASS (exit=$code; expected=$($case.Diagnostic))"
}

# Inspect production dependencies independently of fixture dev-dependencies.
# Cargo can resolve these target trees without installing their Rust std libraries.
foreach ($treeTarget in @('x86_64-unknown-linux-gnu', 'x86_64-pc-windows-msvc')) {
    foreach ($features in @('', 'movable', 'dx12', 'dx12,movable')) {
        $treeArgs = @('tree', '--manifest-path', (Join-Path $stagePath 'Cargo.toml'), '--target', $treeTarget, '--edges', 'normal', '--prefix', 'none')
        if ($features) { $treeArgs += @('--features', $features) }
        $tree = & cargo @treeArgs 2>&1 | Out-String
        if ($LASTEXITCODE -ne 0) { throw "Dependency tree failed: $treeTarget / $features`n$tree" }
        $hasLoader = $tree -match '(?m)^libloading v'
        $expectsLoader = $treeTarget -eq 'x86_64-pc-windows-msvc' -and $features.Split(',').Contains('dx12')
        if ($hasLoader -ne $expectsLoader) { throw "Wrong production loader dependency: $treeTarget / $features`n$tree" }
        Write-Host "production-dependencies-$treeTarget-$features`: PASS (libloading=$hasLoader)"
    }
}

# Check each real package's defaults independently of the consumer's feature wiring.
$packageTrees = @(
    @{ Package='fsr-sdk-sys'; Name='defaults'; CargoArgs=@(); WindowsLoader=$false },
    @{ Package='fsr-sdk-sys'; Name='no-defaults'; CargoArgs=@('--no-default-features'); WindowsLoader=$false },
    @{ Package='fsr-sdk-sys'; Name='dx12'; CargoArgs=@('--no-default-features', '--features', 'dx12'); WindowsLoader=$true },
    @{ Package='fsr-sdk'; Name='defaults'; CargoArgs=@(); WindowsLoader=$true },
    @{ Package='fsr-sdk'; Name='no-defaults'; CargoArgs=@('--no-default-features'); WindowsLoader=$false },
    @{ Package='fsr-sdk'; Name='dx12'; CargoArgs=@('--no-default-features', '--features', 'dx12'); WindowsLoader=$true },
    @{ Package='fsr-sdk'; Name='movable'; CargoArgs=@('--features', 'movable'); WindowsLoader=$true },
    @{ Package='fsr-sdk'; Name='no-defaults-movable'; CargoArgs=@('--no-default-features', '--features', 'movable'); WindowsLoader=$false }
)
foreach ($treeTarget in @('x86_64-unknown-linux-gnu', 'x86_64-pc-windows-msvc')) {
    foreach ($treeConfig in $packageTrees) {
        $treeArgs = @('tree', '--manifest-path', (Join-Path $taskRoot 'Cargo.toml'), '-p', $treeConfig.Package, '--locked', '--target', $treeTarget, '--edges', 'normal', '--prefix', 'none')
        $treeArgs += $treeConfig.CargoArgs
        $tree = & cargo @treeArgs 2>&1 | Out-String
        if ($LASTEXITCODE -ne 0) { throw "Package dependency tree failed: $($treeConfig.Package) / $treeTarget / $($treeConfig.Name)`n$tree" }
        $hasLoader = $tree -match '(?m)^libloading v'
        $expectsLoader = $treeTarget -eq 'x86_64-pc-windows-msvc' -and $treeConfig.WindowsLoader
        if ($hasLoader -ne $expectsLoader) { throw "Wrong package loader dependency: $($treeConfig.Package) / $treeTarget / $($treeConfig.Name)`n$tree" }
        Write-Host "package-dependencies-$($treeConfig.Package)-$treeTarget-$($treeConfig.Name): PASS (libloading=$hasLoader)"
    }
}

# Vulkan remains an explicitly unsupported feature, including on portable hosts.
$vulkan = & cargo check --manifest-path (Join-Path $taskRoot 'crates/fsr-sdk/Cargo.toml') --no-default-features --features vulkan 2>&1 | Out-String
if ($LASTEXITCODE -eq 0 -or !$vulkan.Contains('the `vulkan` feature is intentionally unsupported')) {
    throw "Invalid unsupported-Vulkan control`n$vulkan"
}
Write-Host 'unsupported-vulkan: PASS (expected compile error)'

# All expected compiler failures were validated; report overall success to CI.
exit 0
