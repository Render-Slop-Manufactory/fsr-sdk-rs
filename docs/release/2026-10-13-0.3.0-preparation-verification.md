```python
from pathlib import Path
import sys
sys.path.insert(0, str(Path("tools/verification").resolve()))
from run import parser
from profiles import compose
from execution import Runner
from records import fresh

root = Path.cwd()
options = parser().parse_args([
    "run", "release-artifacts",
    "--wrapper-archive", str(root / "target/package/fsr-sdk-0.3.0.crate"),
    "--sys-archive", str(root / "target/package/fsr-sdk-sys-0.1.2.crate"),
])
cases = [case for case in compose("release-artifacts")
         if ".1.88.0." not in case.id
         and not case.id.startswith("release.consumer-msrv")]
integrity = next(case for case in cases if case.id == "release.integrity")
integrity.requires = tuple(case.id for case in cases if case.id != "release.integrity")
directory = fresh(options.output_root, "draft-packages-current")
runner = Runner(root, directory, options)
runner.state["external_inputs"] = [
    ("wrapper_archive", Path(options.wrapper_archive)),
    ("sys_archive", Path(options.sys_archive)),
]
runner.env["CARGO_TARGET_DIR"] = str(directory / "current-build")
sys.exit(runner.execute(cases, "draft-packages-current"))
```
